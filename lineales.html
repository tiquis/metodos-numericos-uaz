1. Fundamento Teórico

Eliminación Gaussiana

Transforma el sistema \(Ax=b\) en una matriz triangular superior mediante operaciones por renglón. Luego aplica sustitución hacia atrás.

Gauss-Jordan

Extensión del método de Gauss. Elimina incógnitas tanto por encima como por debajo de la diagonal, obteniendo la matriz identidad.

Factorización LU

Descompone \(A\) en una matriz triangular inferior \(L\) y una superior \(U\) (\(A=LU\)). Es ideal cuando se resuelven múltiples vectores \(b\).

Gauss-Seidel

Método iterativo. Despeja \(x_i\) de cada ecuación y actualiza los valores tan pronto como se calculan. Requiere que la matriz sea diagonalmente dominante para converger.

import numpy as np

def eliminacion_gauss(A, b):
    """Resuelve Ax = b usando eliminación Gaussiana simple."""
    n = len(b)
    # Crear matriz aumentada [A|b] (copias para no modificar originales)
    Ab = np.concatenate((A.astype(float), b.astype(float).reshape(n, 1)), axis=1)
    
    # Eliminación hacia adelante
    for k in range(n - 1):
        for i in range(k + 1, n):
            factor = Ab[i, k] / Ab[k, k]
            Ab[i, k:] = Ab[i, k:] - factor * Ab[k, k:]
            
    # Sustitución hacia atrás
    x = np.zeros(n)
    x[n - 1] = Ab[n - 1, n] / Ab[n - 1, n - 1]
    for i in range(n - 2, -1, -1):
        suma = np.dot(Ab[i, i+1:n], x[i+1:n])
        x[i] = (Ab[i, n] - suma) / Ab[i, i]
        
    return x

def gauss_seidel(A, b, tol=1e-5, max_iter=100):
    """Resuelve Ax = b usando el método iterativo de Gauss-Seidel."""
    n = len(b)
    x = np.zeros(n)
    x_old = np.zeros(n)
    
    for k in range(max_iter):
        for i in range(n):
            suma1 = np.dot(A[i, :i], x[:i])       # Valores ya actualizados (k+1)
            suma2 = np.dot(A[i, i+1:], x_old[i+1:]) # Valores viejos (k)
            x[i] = (b[i] - suma1 - suma2) / A[i, i]
            
        # Cálculo del error
        ea = np.max(np.abs((x - x_old) / x)) * 100
        if ea < tol:
            print(f"Convergió en {k+1} iteraciones.")
            return x
            
        x_old = np.copy(x)
        
    print("Alcanzó el máximo de iteraciones sin converger.")
    return x

if __name__ == "__main__":
    A = np.array([[3.0, -0.1, -0.2],
                  [0.1,  7.0, -0.3],
                  [0.3, -0.2, 10.0]])
    b = np.array([7.85, -19.3, 71.4])
    
    x_exact = eliminacion_gauss(A, b)
    print(f"Solución Exacta (Gauss): {x_exact}")
    
    x_iter = gauss_seidel(A, b)
    print(f"Solución Iterativa (Gauss-Seidel): {x_iter}")
