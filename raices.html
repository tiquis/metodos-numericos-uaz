1. Fundamento Teórico

Métodos que usan Intervalos (Bisección)

Requieren dos valores iniciales \(x_l\) y \(x_u\) que "encierren" a la raíz, de forma que \(f(x_l) \cdot f(x_u) < 0\). El método divide el intervalo a la mitad iterativamente. Siempre converge, pero es lento.

$$ x_r = \frac{x_l + x_u}{2} $$

Métodos Abiertos (Newton-Raphson)

Requieren un solo valor inicial \(x_0\). Trazan la recta tangente a la función en \(x_i\) y cruzan el eje X para encontrar el siguiente punto. Convergen muy rápido (cuadráticamente), pero pueden divergir si la derivada es cero o cercana a cero.

$$ x_{i+1} = x_i - \frac{f(x_i)}{f'(x_i)} $$

import math

def biseccion(f, xl, xu, tol=1e-6, max_iter=50):
    """Encuentra la raíz de f(x) mediante el método de bisección."""
    if f(xl) * f(xu) >= 0:
        raise ValueError("El intervalo no encierra una raíz (f(xl) * f(xu) >= 0).")
        
    xr = xl
    xr_old = xl
    
    print(f"{'Iter':<5}{'xl':<12}{'xu':<12}{'xr':<12}{'Ea(%)':<10}")
    
    for i in range(max_iter):
        xr_old = xr
        xr = (xl + xu) / 2
        
        ea = 100.0 if i == 0 else abs((xr - xr_old) / xr) * 100
        
        print(f"{i+1:<5}{xl:<12.6f}{xu:<12.6f}{xr:<12.6f}{ea:<10.4f}")
        
        test = f(xl) * f(xr)
        if test < 0:
            xu = xr
        elif test > 0:
            xl = xr
        else:
            ea = 0.0  # Raíz exacta encontrada
            
        if ea < tol:
            break
            
    return xr

def newton_raphson(f, df, x0, tol=1e-6, max_iter=50):
    """Encuentra la raíz de f(x) dado f'(x) mediante Newton-Raphson."""
    xi = x0
    
    print(f"\n{'Iter':<5}{'xi':<12}{'f(xi)':<12}{f'f\\'(xi)':<12}{'Ea(%)':<10}")
    
    for i in range(max_iter):
        f_val = f(xi)
        df_val = df(xi)
        
        if df_val == 0:
            raise ZeroDivisionError("Derivada cero. El método falla.")
            
        xi_next = xi - f_val / df_val
        ea = abs((xi_next - xi) / xi_next) * 100
        
        print(f"{i+1:<5}{xi:<12.6f}{f_val:<12.6f}{df_val:<12.6f}{ea:<10.4f}")
        
        xi = xi_next
        if ea < tol:
            break
            
    return xi

# --- Ejemplo de Ejecución ---
if __name__ == "__main__":
    # f(x) = e^(-x) - x
    f = lambda x: math.exp(-x) - x
    df = lambda x: -math.exp(-x) - 1
    
    raiz_bis = biseccion(f, xl=0.0, xu=1.0)
    print(f"Raíz por Bisección: {raiz_bis:.6f}")
    
    raiz_nr = newton_raphson(f, df, x0=0.0)
    print(f"Raíz por Newton-Raphson: {raiz_nr:.6f}")
